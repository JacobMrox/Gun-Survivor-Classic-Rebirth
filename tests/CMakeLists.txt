cmake_minimum_required(VERSION 3.20)
project(survivor_fix_regressions LANGUAGES CXX)
enable_testing()
foreach(name IN ITEMS mouse_drain movie_skip)
  add_executable(${name}_test ${name}_test.cpp)
  target_compile_features(${name}_test PRIVATE cxx_std_17)
  add_test(NAME ${name} COMMAND ${name}_test)
endforeach()


add_executable(zombie_audio_test zombie_audio_test.cpp)
target_compile_features(zombie_audio_test PRIVATE cxx_std_17)
add_test(NAME zombie_audio COMMAND zombie_audio_test)

add_executable(title_fade_test title_fade_test.cpp)
target_compile_features(title_fade_test PRIVATE cxx_std_17)
add_test(NAME title_fade COMMAND title_fade_test)

set(renderer_source "${CMAKE_CURRENT_SOURCE_DIR}/../source/survivor_pc.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${renderer_source}")
file(READ "${renderer_source}" renderer_adapter)
string(FIND "${renderer_adapter}" "int SurvivorPc_PresentHeldFrame()" renderer_begin)
string(FIND "${renderer_adapter}" "bool SurvivorPc_BeginTitleTransition()" renderer_end)
if(renderer_begin LESS 0 OR renderer_end LESS renderer_begin)
  message(FATAL_ERROR "The held-frame adapter boundaries were not found")
endif()
math(EXPR renderer_length "${renderer_end} - ${renderer_begin}")
string(SUBSTRING "${renderer_adapter}" ${renderer_begin} ${renderer_length} renderer_adapter)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/held_frame_test.inc" "${renderer_adapter}")
add_executable(held_frame_test held_frame_test.cpp)
target_compile_features(held_frame_test PRIVATE cxx_std_17)
target_include_directories(held_frame_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME held_frame COMMAND held_frame_test)
