cmake_minimum_required(VERSION 3.20)
project(survivor_fix_regressions LANGUAGES CXX)
enable_testing()
foreach(name IN ITEMS mouse_drain movie_skip)
  add_executable(${name}_test ${name}_test.cpp)
  target_compile_features(${name}_test PRIVATE cxx_std_17)
  add_test(NAME ${name} COMMAND ${name}_test)
endforeach()


# Compile the production adapter itself, with a bounded native-query fixture.
set(player_adapter_source "${CMAKE_CURRENT_SOURCE_DIR}/../source/survivor_fix/features/player.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${player_adapter_source}")
file(READ "${player_adapter_source}" player_adapter)
string(FIND "${player_adapter}" "using QueryFn =" adapter_begin)
string(FIND "${player_adapter}" "// Ps1WakeView" adapter_end)
if(adapter_begin LESS 0 OR adapter_end LESS adapter_begin)
  message(FATAL_ERROR "The player collision adapter boundaries were not found")
endif()
math(EXPR adapter_length "${adapter_end} - ${adapter_begin}")
string(SUBSTRING "${player_adapter}" ${adapter_begin} ${adapter_length} player_adapter)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/player_walls_test.inc" "${player_adapter}")
add_executable(player_wall_adapter_test player_wall_adapter_test.cpp
  ../source/survivor_fix/ps1/wall_logic.cpp)
target_compile_features(player_wall_adapter_test PRIVATE cxx_std_17)
target_include_directories(player_wall_adapter_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(player_wall_adapter_test PRIVATE
  SF_TEST_NATIVE_DOOR_ACCESS=1 SF_TEST_STORAGE_DOOR_ACCESS=1)
add_test(NAME player_wall_adapter COMMAND player_wall_adapter_test)
